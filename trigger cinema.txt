-- Trigger cho bảng product_detail
CREATE TRIGGER tg_on_update_price_product
AFTER INSERT OR UPDATE OR DELETE ON product_detail
FOR EACH ROW
EXECUTE FUNCTION tf_on_update_price();

-- Trigger cho bảng seat_detail
CREATE TRIGGER tg_on_update_price_seat
AFTER INSERT OR UPDATE OR DELETE ON seat_detail
FOR EACH ROW
EXECUTE FUNCTION tf_on_update_price();

-- Hàm chung để cập nhật tổng tiền
CREATE OR REPLACE FUNCTION tf_on_update_price()
RETURNS TRIGGER AS
$$
BEGIN
    IF TG_OP = 'INSERT' OR TG_OP = 'UPDATE' THEN
        -- Nếu là thêm mới hoặc cập nhật
        IF TG_TABLE_NAME = 'product_detail' THEN
            -- Xử lý với product_detail
            UPDATE ticket
            SET total_money = total_money 
                + COALESCE(NEW.quantity * NEW.unit_price, 0)
            WHERE ticket_id = NEW.ticket_id;

        ELSIF TG_TABLE_NAME = 'seat_detail' THEN
            -- Xử lý với seat_detail
            UPDATE ticket
            SET total_money = total_money 
                + COALESCE(NEW.price, 0)
            WHERE ticket_id = NEW.ticket_id;
        END IF;

    ELSIF TG_OP = 'DELETE' THEN
        -- Nếu là xóa bản ghi
        IF TG_TABLE_NAME = 'seat_detail' THEN
            -- Xử lý khi xóa từ bảng seat_detail
            UPDATE ticket
            SET total_money = total_money 
                - COALESCE(OLD.price, 0)
            WHERE ticket_id = OLD.ticket_id;

        ELSIF TG_TABLE_NAME = 'product_detail' THEN
            -- Xử lý khi xóa từ bảng product_detail
            UPDATE ticket
            SET total_money = total_money 
                - COALESCE(OLD.quantity * OLD.unit_price, 0)
            WHERE ticket_id = OLD.ticket_id;
        END IF;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;



-- Trigger update giá tiền cho seat_detail
CREATE TRIGGER tg_on_update_seat_price
BEFORE INSERT OR UPDATE ON seat_detail
FOR EACH ROW
EXECUTE PROCEDURE tf_on_update_seat_price();

CREATE OR REPLACE FUNCTION tf_on_update_seat_price()
RETURNS TRIGGER AS
$$
BEGIN
    -- Kiểm tra nếu tồn tại giá phòng cho seat_id được nhập
    IF EXISTS (
        SELECT 1
        FROM schedule
        JOIN room USING (room_id)
	JOIN seat USINg (room_id) 
        WHERE seat_id = NEW.seat_id
    ) THEN
        -- Cập nhật giá tiền từ bảng room
        NEW.price := (
            SELECT room.price
            FROM seat 
            JOIN room USING (room_id)
            WHERE seat_id = NEW.seat_id
        );
    ELSE
        RAISE NOTICE 'seat_id = % not found', NEW.seat_id;
    END IF;

    RETURN NEW; 
END;
$$ LANGUAGE plpgsql;

FOR EACH ROW
EXECUTE PROCEDURE tf_on_update_product_price();

RETURNS TRIGGER AS
$$
BEGIN
    -- Kiểm tra nếu tồn tại product_id được nhập

        SELECT 1
        FROM product  
        WHERE product_id = NEW.product_id

    END IF;

    RETURN NEW; 
END;
$$ LANGUAGE plpgsql;


-- Trigger xử lí discount
CREATE TRIGGER tg_on_discount
AFTER UPDATE ON ticket
FOR EACH ROW
WHEN (NEW.discount_id IS DISTINCT FROM OLD.discount_id)
EXECUTE PROCEDURE tf_on_discount();

CREATE OR REPLACE FUNCTION tf_on_discount()
RETURNS TRIGGER AS
$$
BEGIN
    IF NEW.discount_id IS NOT NULL THEN
        IF EXISTS (
            SELECT 1
            FROM discount
            WHERE discount_id = NEW.discount_id
            AND date_expire < CURRENT_DATE
        ) THEN
            RAISE NOTICE 'discount_id = % is expired!', NEW.discount_id;
            RETURN NEW;
        END IF;
   	IF EXISTS (
            SELECT 1
            FROM discount
            WHERE discount_id = NEW.discount_id
        ) THEN
            UPDATE ticket
            SET total_money = OLD.total_money * (1 - (
                SELECT value::float / 100
                FROM discount
                WHERE discount_id = NEW.discount_id
            ))
            WHERE ticket_id = OLD.ticket_id;
        ELSE
            RAISE NOTICE 'discount_id = % does not exist!', NEW.discount_id;
        END IF;

    ELSIF OLD.discount_id IS NOT NULL THEN
        UPDATE ticket
        SET total_money = OLD.total_money / (1 - (
            SELECT value::float / 100
            FROM discount
            WHERE discount_id = OLD.discount_id
        ))
        WHERE ticket_id = OLD.ticket_id;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;


-- Trigger update tổng spending
CREATE TRIGGER tg_insert_on_spending
AFTER UPDATE ON ticket
FOR EACH ROW
WHEN (NEW.total_money IS DISTINCT FROM OLD.total_money)
EXECUTE PROCEDURE tf_update_on_spending();

CREATE OR REPLACE FUNCTION tf_update_on_spending()
RETURNS TRIGGER AS
$$
BEGIN
	UPDATE users
	SET total_amount = (SELECT sum(total_money) FROM ticket WHERE user_id = COALESCE(NEW.user_id, OLD.user_id)) WHERE user_id = COALESCE(NEW.user_id, OLD.user_id);
RETURN NEW;
END;
$$ LANGUAGE plpgsql;


-- Trigger khi thay đổi user_id trong ticket 
CREATE TRIGGER tg_update_user_id
AFTER UPDATE OF user_id ON ticket
FOR EACH ROW
EXECUTE PROCEDURE tf_on_update_userid_money();

CREATE OR REPLACE FUNCTION tf_on_update_userid_money()
RETURNS TRIGGER AS
$$
BEGIN
	UPDATE users
	SET total_amount = total_amount - (SELECT total_money FROM ticket WHERE ticket_id = OLD.ticket_id)
	WHERE user_id = OLD.user_id;

	UPDATE users
	SET total_amount = total_amount + (SELECT total_money FROM ticket WHERE ticket_id = NEW.ticket_id)
	WHERE user_id = NEW.user_id;

RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Trigger delete on ticket;

after delete on ticket

returns trigger as

begin
	update users 
	set total_amount = total_amount - old.total_money where user_id = old.user_id;
return new;
end;
$$

before insert or update of schedule_id on ticket
for each row

returns trigger as
$$
declare age_restriction int;
begin
	age_restriction = (select ageres from movie m join schedule using (movie_id) where schedule_id = new.schedule_id);
	raise notice 'This movie has the age restrict is % please notice this!', age_restriction;
return new;
end;
$$ language plpgsql;
